// --- Пользователи ---
Table users {
  id uuid [pk, default: `gen_random_uuid()`]
  username varchar
  email varchar [not null, unique]
  first_name varchar
  last_name varchar
  whatsapp bigint [not null]
  telegram varchar
  address_id int [ref: > addresses.id, null] // адрес из профиля (может быть null)
  referral_code varchar [unique, not null]
  referred_by_code varchar
  role varchar [not null, default: 'user'] // user, admin, operator
  created_at timestamp [default: `now()`]
}

// --- Адреса ---
Table addresses {
  id int [pk, increment]

  country varchar
  city varchar
  street varchar
  house_number varchar
  postal_code varchar

  full_address text [not null]
  latitude decimal
  longitude decimal

  created_at timestamp [default: `now()`]
}
// --- Филиалы ---
Table branches {
  id int [pk, increment]
  name varchar [not null]
  email varchar [unique]
  whatsapp bigint
  telegram varchar
  instagram varchar
  address_id int [ref: > addresses.id, not null] // физический адрес филиала
  is_active boolean [default: true]
  created_at timestamp [default: `now()`]
}

// --- График работы филиалов ---
Table branch_hours {
  id int [pk, increment]
  branch_id int [ref: > branches.id]
  weekday int [not null] // 0 = Sunday, 1 = Monday, ..., 6 = Saturday
  open_time time
  close_time time
  is_open boolean [default: true]

  Indexes {
    (branch_id, weekday) [unique]
  }
}

// --- Валюты ---
Table currencies {
  id int [pk, increment]
  code varchar [unique, not null] // USD, EUR, ARS
  name varchar [unique, not null]
  symbol varchar
  decimal_places int [default: 2]
  created_at timestamp [default: `now()`]
}

// --- Валюты, привязанные к филиалам ---
Table branch_currencies {
  id int [pk, increment]
  branch_id int [ref: > branches.id]
  currency_id int [ref: > currencies.id]
  amount decimal [not null, default: 0]

  Indexes {
    (branch_id, currency_id) [unique]
  }
}

// --- Курсы валют ---
Table exchange_rates {
  id int [pk, increment]
  currency_from_id int [ref: > currencies.id]
  currency_to_id int [ref: > currencies.id]
  branch_id int [ref: > branches.id]
  min_amount decimal [not null]
  max_amount decimal
  rate decimal [not null]
  is_hot boolean [default: false]
  visible boolean [default: true]
  in_filter boolean [default: false]
  updated_at timestamp [default: `now()`]

  Indexes {
    (currency_from_id, currency_to_id, branch_id, min_amount) [unique]
  }
}

// --- Заказы ---
Table orders {
  id uuid [pk, increment]
  user_id uuid [ref: > users.id]
  branch_id int [ref: > branches.id]
  currency_from_id int [ref: > currencies.id]
  currency_to_id int [ref: > currencies.id]
  amount_from decimal [not null]
  amount_to decimal [not null]
  applied_rate decimal [not null]
  status varchar [not null, default: 'pending'] // pending, completed, canceled
  delivery boolean [default: false]
  delivery_address_id int [ref: > addresses.id, null] // если выбрана доставка  created_at timestamp [default: `now()`]
  completed_at timestamp
}

// --- Отзывы ---
Table reviews {
  id int [pk, increment]
  user_id uuid [ref: > users.id]
  order_id uuid [ref: > orders.id]
  rating int [not null] // 1–5
  comment text
  visible boolean [default: false]
  created_at timestamp [default: `now()`]

  Indexes {
    (user_id, order_id) [unique]
  }
}

Table referral_transactions {
  id int [pk, increment]
  user_id uuid [ref: > users.id] // кто получает
  referred_user_id uuid [ref: > users.id] // кто привёл доход
  order_id int [ref: > orders.id]
  amount decimal [not null]
  created_at timestamp [default: `now()`]
}
