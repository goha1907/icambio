// --- Пользователи ---
Table users {
  id uuid [pk, default: `gen_random_uuid()`]
  username varchar
  email varchar [not null, unique]
  first_name varchar
  last_name varchar
  whatsapp bigint
  telegram varchar
  referral_code varchar [unique, not null]
  referred_by_code varchar
  referral_amount decimal [not null, default: 0]
  address_id int [ref: > addresses.id, null]
  role varchar [not null, default: 'user']
  created_at timestamp [default: `now()`]
}

// --- Адреса ---
Table addresses {
  id int [pk, increment]
  country varchar
  city varchar
  street varchar
  house_number varchar
  postal_code varchar
  full_address text
  latitude decimal
  longitude decimal
  created_at timestamp [default: `now()`]
}

// --- Филиалы ---
Table branches {
  id int [pk, increment]
  name varchar [not null]
  email varchar
  whatsapp bigint
  telegram varchar
  instagram varchar
  address_id int [ref: > addresses.id]
  is_active boolean [default: true]
  created_at timestamp [default: `now()`]
}

// --- График работы филиала ---
Table branch_hours {
  id int [pk, increment]
  branch_id int [ref: > branches.id]
  weekday int [not null]
  is_open boolean [default: true]
  open_time time
  close_time time

  Indexes {
    (branch_id, weekday) [unique]
  }
}

// --- Валюты ---
Table currencies {
  id int [pk, increment]
  code varchar [unique, not null]
  name varchar [unique, not null]
  symbol varchar
  decimal_places int [default: 2]
  created_at timestamp [default: `now()`]
}

// --- Резервы валют ---
Table branch_currencies {
  id int [pk, increment]
  branch_id int [ref: > branches.id]
  currency_id int [ref: > currencies.id]
  amount decimal [not null, default: 0]

  Indexes {
    (branch_id, currency_id) [unique]
  }
}

// --- Курсы обмена ---
Table exchange_rates {
  id int [pk, increment]
  branch_id int [ref: > branches.id]
  currency_from_id int [ref: > currencies.id]
  currency_to_id int [ref: > currencies.id]
  min_amount decimal [not null]
  max_amount decimal [null]
  rate decimal [not null]
  is_hot boolean [default: false]
  visible boolean [default: false]
  in_filter boolean [default: false]
  updated_at timestamp [default: `now()`]

  Indexes {
    (currency_from_id, currency_to_id, branch_id, min_amount) [unique]
  }
}

// --- Заказы ---
Table orders {
  id int [pk, increment]
  user_id uuid [ref: > users.id]
  branch_id int [ref: > branches.id]
  currency_from_id int [ref: > currencies.id]
  currency_to_id int [ref: > currencies.id]
  amount_from decimal [not null]
  amount_to decimal [not null]
  applied_rate decimal [not null]
  status varchar [not null, default: 'pending']
  delivery boolean [default: false]
  delivery_address_id int [ref: > addresses.id, null]
  created_at timestamp [default: `now()`]
  completed_at timestamp
}

// --- Прибыль по заказу ---
Table order_profit {
  id int [pk, increment]
  order_id int [ref: > orders.id]
  currency_id int [ref: > currencies.id]
  amount decimal [not null]
  created_at timestamp [default: `now()`]
}

// --- Отзывы ---
Table reviews {
  id int [pk, increment]
  user_id uuid [ref: > users.id]
  order_id int [ref: > orders.id]
  rating int [not null]
  comment text
  visible boolean [default: false]
  created_at timestamp [default: `now()`]

  Indexes {
    (user_id, order_id) [unique]
  }
}

// --- Реферальные операции ---
Table referral_transactions {
  id int [pk, increment]
  user_id uuid [ref: > users.id]
  referred_user_id uuid [ref: > users.id, null]
  order_id int [ref: > orders.id, null]
  type varchar [not null] // earn, withdrawal, adjustment
  amount decimal [not null]
  currency_id int [ref: > currencies.id]
  branch_id int [ref: > branches.id, null]
  note text
  created_at timestamp [default: `now()`]
}

// --- Закупки валют ---
Table purchases {
  id int [pk, increment]
  branch_id int [ref: > branches.id]
  currency_from_id int [ref: > currencies.id]
  currency_to_id int [ref: > currencies.id]
  amount_from decimal [not null]
  amount_to decimal [not null]
  applied_rate decimal [not null]
  source varchar
  user_id uuid [ref: > users.id, null]
  note text
  created_at timestamp [default: `now()`]
}

// --- Движения валют ---
Table currency_movements {
  id int [pk, increment]
  branch_id int [ref: > branches.id]
  currency_id int [ref: > currencies.id]
  user_id uuid [ref: > users.id, null]
  order_id int [ref: > orders.id, null]
  purchase_id int [ref: > purchases.id, null]
  referral_transaction_id int [ref: > referral_transactions.id, null]
  type varchar [not null] // deposit, withdrawal, correction
  amount decimal [not null]
  reason text
  created_at timestamp [default: `now()`]
}
